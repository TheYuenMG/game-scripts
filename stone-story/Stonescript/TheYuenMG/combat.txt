// *** Ability Activation ***
func CanUse(weapon)
    return item.CanActivate() &
    ^item.GetCooldown(weapon) <= 0

func Use(weapon)
    equip @weapon@
    activate R


// *** Attack Animation Cancelling ***
?item.left.state = 3
    equipL *0 wand //trash
    equipL @item.left@

?item.right.state = 3
    equipR *0 shield //trash
    equip @item.right@
