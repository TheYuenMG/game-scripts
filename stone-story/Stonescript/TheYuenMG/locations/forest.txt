?((foe = ant_hill & foe.distance <= 18) |
^screen.i >= 4) & item.CanActivate() &
^item.potion = exp
    activate potion

?pickup.distance < 10
    equipL star

:?foe.distance > 18
    equipL trisk
    equip poi shield

:?screen.i >= 5 & item.CanActivate() &
^item.GetCooldown(blade) <= 0
    equip Blade of
    activate R

:?foe = boss

    ?(item.GetCooldown(heavy_hammer) <= 0 &
    ^item.CanActivate()) |
    ^(item.right = heavy hammer &
    ^item.right.state = 2)
        equip heavy hammer
        activate R

    :
        equipL poi hammer
        equip poi shield

:
    equipL +0 vig wand
    equipR vig wand
