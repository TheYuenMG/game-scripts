?foe.distance <= 5 & item.potion = exp &
^item.CanActivate()
    activate potion

?pickup.distance < 10
    equipR star

:?foe.distance > 18
    equipL trisk

:?foe = boss
    equip ice sword

    ?item.GetCooldown("bardiche") <= 0 &
    ^foe.distance < 5 &
    ^item.CanActivate() |
    ^(item.right = "bardiche" &
    ^item.right.state = 2)
        equip bard
        activate R

:
    // just kidding repeater sucks
    equipR wand
