?loc = caves
    ?hp <= maxhp * 0.2 & item.potion = heal
        activate potion

    ?foe.distance <= 18
        equipL exp
        equipR vig sword

    ?foe = boss
        equipL vig sword
        equip com shield
        ?foe.hp <= foe.maxhp * 0.1
            equipR exp

    ?foe.distance > 18
        equipL trisk

    ?pickup.distance < 10
        equipR star
