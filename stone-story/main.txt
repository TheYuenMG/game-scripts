import Pets/Frog
import UI/MindstoneButton


// *** Attack Animation Cancelling ***
?item.left.state = 3
    equipL *0 wand //trash
    equipL @item.left@

?item.right.state = 3
    equipR *0 shield //trash
    equip @item.right@


// *** Deadwood Experience Farming ***
?loc = deadwood
    ?foe.distance <= 5 & item.potion = exp
        activate potion

    ?foe.distance <= 18
        equipL exp
        equipR vig sword

    ?foe = boss
        equipL vig sword
        equip com shield
        ?foe.hp <= foe.maxhp * 0.1
            equipR exp

    ?foe.distance > 18
        equipL trisk

    ?pickup.distance < 10
        equipR star

    ?harvest.distance < 10
        equip hatchet


// *** Caves Tar Farming ***
?loc = caves
    ?hp <= maxhp * 0.2 & item.potion = heal
        activate potion

    ?foe.distance <= 18
        equipL exp
        equipR vig sword

    ?foe = boss
        equipL vig sword
        equip com shield
        ?foe.hp <= foe.maxhp * 0.1
            equipR exp

    ?foe.distance > 18
        equipL trisk

    ?pickup.distance < 10
        equipR star
