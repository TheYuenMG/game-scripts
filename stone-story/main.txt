// *** Community Scripts ***
import Fishing
import Hats/Skully
import Pets/Boo
import Pets/Frog
import UI/FoeStateTracker
import UI/MindstoneButton


// *** Attack Animation Cancelling ***
?item.left.state = 3
    equipL *0 wand //trash
    equipL @item.left@

?item.right.state = 3
    equipR *0 shield //trash
    equip @item.right@


// *** Deadwood Wood Farming ***
?loc = deadwood
    ?foe.distance <= 5 & item.potion = exp
        activate potion

    ?harvest.distance < 10
        equip hatchet

        ?item.GetCooldown(hatchet) <= 0 &
        ^item.CanActivate()
            activate R

    :?pickup.distance < 10
        equipL star

    :?foe.distance > 18
        equipL trisk
        equip com shield

    :?foe = boss & foe.distance <= 10

        ?(item.GetCooldown(bardiche) <= 0 &
        ^item.CanActivate()) |
        ^(item.right = bardiche &
        ^item.right.state = 2)
            equip bardiche
            activate R

        :
            equipL vig sword
            equip com shield

    :
        equipL ki
        equipR vig sword


// *** Forest Vigor Rune Farming ***
?loc = forest
    ?foe = ant_hill & foe.distance <= 18 &
    ^item.potion = exp
        activate potion

    ?pickup.distance < 10
        equipL star

    :?foe.distance > 18
        equipL trisk
        equip poi shield

    :?foe = boss

        ?(item.GetCooldown(heavy_hammer) <= 0 &
        ^item.CanActivate()) |
        ^(item.right = heavy hammer &
        ^item.right.state = 2)
            equip heavy hammer
            activate R

        :
            equipL poi hammer
            equip poi shield

    :
        equipL *9 vig wand
        equipR *10 vig wand


// *** Plateau Event Farming ***
?loc = plateau
    ?hp <= maxhp * 0.2 & item.potion = heal
        activate potion

    ?harvest.distance < 10
        equip shovel

        ?item.GetCooldown(shovel) <= 0 &
        ^item.CanActivate()
            activate R

    :?pickup.distance < 10
        equipL star

    :?foe.distance > 18
        equipL trisk
        equip com shield

    :
        equipL vig sword
        equip com shield


>Warm Stones: @item.GetCount(warm_stone)@
