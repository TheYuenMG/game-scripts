// *** Community Scripts ***
import Fishing
import Hats/Skully
import Pets/Boo
import Pets/Frog
import UI/MindstoneButton


// *** Attack Animation Cancelling ***
?item.left.state = 3
    equipL *0 wand //trash
    equipL @item.left@

?item.right.state = 3
    equipR *0 shield //trash
    equip @item.right@


// *** Crypt Blade of the Fallen God R&D ***
?loc = crypt
    ?foe.distance <= 18 & item.potion = exp
        activate potion

    ?foe.distance <= 18
        equipL ki
        equipR vig wand

        // wait for R.I.P. to spawn Smite fodder
        ?screen.i = 5 &
        // avoid being a squatting duck, though
        ^item.GetCooldown(＂blade＂) <= 0
            equipL quest
            equip com shield

            // 10 at a time should be enough,
            // adjust number as needed
            ?item.GetCooldown(＂blade＂) <= 0 &
            ^foe.count >= 10
                equip Blade of
                activate R

    ?foe = boss
        equipL vig wand
        equip com shield

        // Save the Date for good ol' Pally-Wally!
        ?buffs.GetCount(＂smite＂) > 0
            /*
            - Rune Sword for fast attack speed
            - Ice because Chill debuff stacks,
            unlike Burn or Poison for some reason
            - *0 to ensure all Smite stacks used
            */
            equipL *0 ice sword

    ?foe.distance > 18
        equipL trisk


// *** Deadwood Wood Farming ***
?loc = deadwood
    ?foe.distance <= 5 & item.potion = exp
        activate potion

    ?foe.distance <= 18
        equipL ki
        equipR vig sword

    ?foe = boss
        equipL vig sword
        equip com shield

    ?foe.distance > 18
        equipL trisk

    ?pickup.distance < 10
        equipR star

    ?harvest.distance < 10
        equip hatchet
