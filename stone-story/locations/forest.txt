?loc = forest
    ?foe = ant_hill & foe.distance <= 18 &
    ^item.potion = exp
        activate potion

    ?pickup.distance < 10
        equipL star

    :?foe.distance > 18
        equipL trisk
        equip poi shield

    :?foe = boss

        ?(item.GetCooldown(heavy_hammer) <= 0 &
        ^item.CanActivate()) |
        ^(item.right = heavy hammer &
        ^item.right.state = 2)
            equip heavy hammer
            activate R

        :
            equipL poi hammer
            equip poi shield

    :
        equipL *9 vig wand
        equipR *10 vig wand
