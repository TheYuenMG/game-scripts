    ?foe.distance <= 18

        equipR vig sword

        /*

        ?item.GetCooldown("blade") <= 0 &
        ^foe.count >= 5
        // to Smite groups of creatures at once
            equip Blade of
            activate R

        // mutated runes for Vigor Sword, Stun is
        // inefficient but so is re-mutation
        ?buffs.GetCount("smite") > 0
            equip grappling
            equip com shield

    ?foe = boss
        equipL vig sword
        equip com shield
        ?foe.hp <= foe.maxhp * 0.1
            equipR exp

    ?foe.distance > 18
        equipL trisk
