?loc = deadwood
    ?foe.distance <= 5 & item.potion = exp
        activate potion

    ?foe.distance <= 18
        equipL exp
        equipR vig sword

    ?foe = boss
        equipL vig sword
        equip com shield
        ?foe.hp <= foe.maxhp * 0.1
            equipR exp

    ?foe.distance > 18
        equipL trisk

    ?pickup.distance < 10
        equipR star

    ?harvest.distance < 10
        equip hatchet
