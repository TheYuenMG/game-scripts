?loc = plateau
    ?hp <= maxhp * 0.2 & item.potion = heal
        activate potion

    ?harvest.distance < 10
        equip shovel

        ?item.GetCooldown(shovel) <= 0 &
        ^item.CanActivate()
            activate R

    :?pickup.distance < 10
        equipL star

    :?foe.distance > 18
        equipL trisk
        equip com shield

    :
        equipL vig sword
        equip com shield
